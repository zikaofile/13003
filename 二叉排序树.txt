#include <stdio.h>
#include<stdlib.h>


//1.定义树节点
typedef struct node
{
	int key;
	struct node *lchild,*rchild;
}BSTNode,*BSTree;


//2.函数声明
BSTree CreateBST(void);
void InsertBST(BSTree T,int Key);     //往树中插入一个结点
void DelBSTNode(BSTree T,int Key);    //删除一个结点
void InoderBST(BSTree T);     //排序


//3.主程序
void main()
{
	BSTree T;
	char ch1,ch2;
	int Key;
	printf("建立一颗二叉排序树的二叉链表存储\n");
	T=CreateBST();
	ch1='y';
	while(ch1=='y'||ch1=='Y')
	{
		printf("请选择下列操作:\n");
		printf("A-------更新二叉排序树存储\n");
		printf("B-------二叉排序树上的删除\n");
		printf("C-------二叉排序树中序输出\n");
		printf("D-------退出\n");
		scanf("%c",&ch2);
		switch(ch2)
		{
		case 'A':
		case 'a':
			{
				T=CreateBST();
				break;
			}
		case 'B':
		case 'b':
			{
				printf("\n请输入要删除的数据:");
				scanf("\n%d",&Key);
				DelBSTNode(T,Key);
				printf("删除成功\n");
				break;
			}
		case 'C':
		case 'c':
			{
				InoderBST(T);
				printf("二叉排序树输出完毕\n");
				break;
			}
		case 'D':
		case 'd':
			{
				ch1='n';
				break;
			}
		default: ch1='n';
		}

	}
}
//4. 函数的实现
BSTree CreateBST(void)
{
	BSTree T;
	int Key;
	T=NULL;
	printf("请输入一个关键字，输入0时结束输入：\n");
	scanf("%d",&Key);
	while(Key)
	{
		InsertBST(T,Key);// 考生填写
		printf("请输入下一个关键字，输入0时结束输入：\n");
		scanf("%d",&Key);
	}
	return T;
}

void InsertBST(BSTree T,int Key)
{
	BSTree f,p;
	p=T;
	while(p)
	{
		if(p->key==Key)
		{
			printf("树中已有key，不需要插入\n");
			return;
		}
		f=p;
		p=(Key<p->key)?p->lchild:p->rchild;		
	}
	p=(BSTree)malloc(sizeof(BSTNode));
	p->key=Key;
	p->lchild=p->rchild=NULL;
	if(T==NULL)
	{
		T=p;
	}
	else if(Key<f->key)
	{
		f->lchild=p;
	}
	else
	{
		f->rchild=p;
	}

}

void DelBSTNode(BSTree T,int Key)
{
	BSTree parent =NULL,p,q,child;
	p=T;
	while(p)
	{
		if(p->key==Key)
			break;
		parent =p;
		p=(Key<p->key)?p->lchild:p->rchild;
	}
	if(!p)
	{

		printf("没有找到要删除的节点\n");
		return;
	}
	q=p;
	if(q->lchild &&	q->rchild)
	{
		for(parent=q,p=q->rchild;p->lchild;parent=p,p=p->lchild);
		child=(p->lchild)?p->lchild:p->rchild;
	}
	if(!parent)
		T=child;
	else
	{
		if(p==parent->lchild)
			parent->lchild=child;
		else
			parent->rchild=child;
		if(p!=q)
				q->key=p->key;
	
	}
		free(p);
}


void InoderBST(BSTree T)
{
	if(T!=NULL)
	{
		InoderBST(T->lchild);
		printf("%d",T->key);
		InoderBST(T->rchild);
	}
}
